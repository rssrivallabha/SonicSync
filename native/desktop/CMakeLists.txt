cmake_minimum_required(VERSION 3.20)
project(SonicSyncDesktopNative VERSION 10.0.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(sonicsync-native src/main.cpp src/native_audio.cpp src/local_transport.cpp)
target_include_directories(sonicsync-native PRIVATE include)
if(WIN32)
  target_link_libraries(sonicsync-native PRIVATE ole32 avrt mmdevapi)
elseif(APPLE)
  find_library(AUDIO_TOOLBOX AudioToolbox REQUIRED)
  find_library(AUDIO_UNIT AudioUnit REQUIRED)
  find_library(CORE_AUDIO CoreAudio REQUIRED)
  target_link_libraries(sonicsync-native PRIVATE ${AUDIO_TOOLBOX} ${AUDIO_UNIT} ${CORE_AUDIO})
elseif(UNIX)
  find_package(PkgConfig)
  if(PKG_CONFIG_FOUND)
    pkg_check_modules(PIPEWIRE libpipewire-0.3 QUIET)
  endif()
  if(PIPEWIRE_FOUND)
    target_compile_definitions(sonicsync-native PRIVATE SONICSYNC_HAS_PIPEWIRE=1)
    target_include_directories(sonicsync-native PRIVATE ${PIPEWIRE_INCLUDE_DIRS})
    target_link_directories(sonicsync-native PRIVATE ${PIPEWIRE_LIBRARY_DIRS})
    target_link_libraries(sonicsync-native PRIVATE ${PIPEWIRE_LIBRARIES})
  else()
    target_compile_definitions(sonicsync-native PRIVATE SONICSYNC_HAS_PIPEWIRE=0)
    message(WARNING "PipeWire development package not installed: Linux backend is source-complete but not linked in this build.")
  endif()
endif()
